JSON.parse(window.localStorage.getItem('netlify-cms-user')).token : null; if (!token) throw new Error('Not authenticated'); const presignResp = await fetch('https://sudd-library-auth.anznexus.workers.dev/media/presign', { method: 'POST', headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${token}`, }, body: JSON.stringify({ filename: file.name, contentType: file.type }), }); if (!presignResp.ok) throw new Error('Failed to get upload URL'); const { uploadUrl, publicUrl } = await presignResp.json(); const uploadResp = await fetch(uploadUrl, { method: 'PUT', headers: { 'Content-Type': file.type }, body: file, }); if (!uploadResp.ok) throw new Error('Upload to R2 failed'); return { url: publicUrl }; }, });
